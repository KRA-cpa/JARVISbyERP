function extractPayload(data, requiredFields = []) {
  // Handle both { action, payload } and { action, ...fields } formats
  let payload = data.payload;

  if (!payload) {
    // Extract from root level (spread format)
    payload = {};
    requiredFields.forEach(field => {
      if (data[field] !== undefined) {
        payload[field] = data[field];
      }
    });
  }

  // ✅ PROPER VALIDATION: Check payload exists before accessing properties
  if (!payload) {
    throw new Error('Payload is required');
  }

  return payload;
}

function createCompany(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let companiesSheet = ss.getSheetByName(SHEETS.COMPANIES);

    if (!companiesSheet) {
      companiesSheet = initializeCompaniesSheet();
    }

    // Check if code already exists
    const existingCompanies = getSheetDataAsJSON(companiesSheet);
    const codeExists = existingCompanies.some(c =>
      c.code && c.code.toUpperCase() === payload.code.toUpperCase()
    );

    if (codeExists) {
      throw new Error('Company code already exists');
    }

    // Generate new company
    const newCompany = {
      id: generateId('comp'),
      name: payload.name.trim(),
      code: payload.code.toUpperCase().trim(),
      created_at: new Date().toISOString(),
      updated_at: new Date().toISOString()
    };

    // Add to sheet
    companiesSheet.appendRow([
      newCompany.id,
      newCompany.name,
      newCompany.code,
      newCompany.created_at,
      newCompany.updated_at
    ]);

    Logger.log(`Created company: ${newCompany.name} (${newCompany.code})`);
    return newCompany;
  } catch (error) {
    Logger.log(`Error creating company: ${error.message}`);
    throw error;
  }
}

function updateCompany(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const companiesSheet = ss.getSheetByName(SHEETS.COMPANIES);

    // Find the row to update
    const data = companiesSheet.getDataRange().getValues();
    const headers = data[0];
    let rowIndex = -1;

    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === payload.id) {
        rowIndex = i + 1;
        break;
      }
    }

    if (rowIndex === -1) {
      throw new Error('Company not found');
    }

    // Check if code already exists (excluding current company)
    const existingCompanies = getSheetDataAsJSON(companiesSheet);
    const codeExists = existingCompanies.some(c =>
      c.id !== payload.id && c.code && c.code.toUpperCase() === payload.code.toUpperCase()
    );

    if (codeExists) {
      throw new Error('Company code already exists');
    }

    // Update the company
    const updatedCompany = {
      id: payload.id,
      name: payload.name.trim(),
      code: payload.code.toUpperCase().trim(),
      created_at: data[rowIndex - 1][3],
      updated_at: new Date().toISOString()
    };

    // Update the row
    companiesSheet.getRange(rowIndex, 1, 1, 5).setValues([[
      updatedCompany.id,
      updatedCompany.name,
      updatedCompany.code,
      updatedCompany.created_at,
      updatedCompany.updated_at
    ]]);

    Logger.log(`Updated company: ${updatedCompany.name}`);
    return updatedCompany;
  } catch (error) {
    Logger.log(`Error updating company: ${error.message}`);
    throw error;
  }
}

function createTicket(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let ticketsSheet = ss.getSheetByName(SHEETS.TICKETS);

    if (!ticketsSheet) {
      ticketsSheet = initializeTicketsSheet();
    }

    const now = new Date();
    const newTicketId = generateId('ticket');

    // Generate ticket number if company and type info available
    let ticketNumber = newTicketId; // Fallback
    if (payload.company_id && payload.ticket_type_id) {
      try {
        const companies = getSheetDataAsJSON(ss.getSheetByName(SHEETS.COMPANIES));
        const ticketTypes = getSheetDataAsJSON(ss.getSheetByName(SHEETS.TICKET_TYPES));

        const company = companies.find(c => c.id === payload.company_id);
        const ticketType = ticketTypes.find(t => t.id === payload.ticket_type_id);

        if (company && ticketType) {
          ticketNumber = generateTicketNumber(company.code, ticketType.code);
        }
      } catch (error) {
        Logger.log('Could not generate formatted ticket number:', error.message);
      }
    }

    // Create ticket entry
    const newTicket = {
      id: newTicketId,
      ticket_number: ticketNumber,
      title: payload.title,
      ticket_type_id: payload.ticket_type_id || '',
      requester_id: payload.requester_id || '',
      status: payload.status || 'New',
      current_step_id: payload.current_step_id || '',
      step_due_date: payload.step_due_date || '',
      created_at: now.toISOString(),
      updated_at: now.toISOString(),
      company_id: payload.company_id || ''
    };

    // Add to sheet
    ticketsSheet.appendRow([
      newTicket.id,
      newTicket.ticket_number,
      newTicket.title,
      newTicket.ticket_type_id,
      newTicket.requester_id,
      newTicket.status,
      newTicket.current_step_id,
      newTicket.step_due_date,
      newTicket.created_at,
      newTicket.updated_at,
      newTicket.company_id
    ]);

    Logger.log(`Created ticket: ${newTicket.ticket_number}`);
    return newTicket;
  } catch (error) {
    Logger.log(`Error creating ticket: ${error.message}`);
    throw error;
  }
}

function updateTicket(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const ticketsSheet = ss.getSheetByName(SHEETS.TICKETS);

    // Find the row to update
    const data = ticketsSheet.getDataRange().getValues();

    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === payload.id) {
        rowIndex = i + 1;
        break;
      }
    }

    if (rowIndex === -1) {
      throw new Error('Ticket not found');
    }

    // Update ticket fields
    const updatedTicket = {
      id: payload.id,
      ticket_number: data[rowIndex - 1][1], // Keep original ticket number
      title: payload.title || data[rowIndex - 1][2],
      ticket_type_id: payload.ticket_type_id || data[rowIndex - 1][3],
      requester_id: payload.requester_id || data[rowIndex - 1][4],
      status: payload.status || data[rowIndex - 1][5],
      current_step_id: payload.current_step_id || data[rowIndex - 1][6],
      step_due_date: payload.step_due_date || data[rowIndex - 1][7],
      created_at: data[rowIndex - 1][8], // Keep original created_at
      updated_at: new Date().toISOString(),
      company_id: payload.company_id || data[rowIndex - 1][10]
    };

    // Update the row
    ticketsSheet.getRange(rowIndex, 1, 1, 11).setValues([[
      updatedTicket.id,

    let optionsSheet = ss.getSheetByName(SHEETS.DROPDOWN_OPTIONS);

    if (!listsSheet) {
      listsSheet = initializeDropdownListsSheet();
    }
    if (!optionsSheet) {
      optionsSheet = initializeDropdownOptionsSheet();
    }

    const newList = {
      id: generateId('dd'),
      name: payload.name.trim(),
      created_at: new Date().toISOString(),

    listsSheet.appendRow([

      newList.created_at,
      newList.updated_at
    ]);

    // Add options if provided
    if (payload.options && Array.isArray(payload.options)) {
      payload.options.forEach(option => {
        if (option.label && option.value) {
          optionsSheet.appendRow([
            generateId('opt'),
            newList.id,
            option.label.trim(),
            option.value.trim(),
            option.parentValue || ''
          ]);
        }
      });
    }

    Logger.log(`Created dropdown list: ${newList.name}`);
    return newList;
  } catch (error) {
    Logger.log(`Error creating dropdown list: ${error.message}`);
    throw error;
  }
}

function updateDropdownList(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const listsSheet = ss.getSheetByName(SHEETS.DROPDOWN_LISTS);
    const optionsSheet = ss.getSheetByName(SHEETS.DROPDOWN_OPTIONS);

    // Update the list
    const listData = listsSheet.getDataRange().getValues();
    let listRowIndex = -1;

    for (let i = 1; i < listData.length; i++) {
      if (listData[i][0] === payload.id) {
        listRowIndex = i + 1;
        break;
      }
    }

    if (listRowIndex === -1) {
      throw new Error('Dropdown list not found');
    }

    const updatedList = {
      id: payload.id,
      name: payload.name.trim(),
      created_at: listData[listRowIndex - 1][2],
      updated_at: new Date().toISOString()
    };

    listsSheet.getRange(listRowIndex, 1, 1, 4).setValues([[
      updatedList.id,
      updatedList.name,
      updatedList.created_at,
      updatedList.updated_at
    ]]);

    // Delete existing options
    const optionsData = optionsSheet.getDataRange().getValues();
    for (let i = optionsData.length - 1; i >= 1; i--) {
      if (optionsData[i][1] === payload.id) {
        optionsSheet.deleteRow(i + 1);
      }
    }

    // Add new options
    if (payload.options && Array.isArray(payload.options)) {
      payload.options.forEach(option => {
        if (option.label && option.value) {
          optionsSheet.appendRow([
            generateId('opt'),
            payload.id,
            option.label.trim(),
            option.value.trim(),
            option.parentValue || ''
          ]);
        }
      });
    }

    Logger.log(`Updated dropdown list: ${updatedList.name}`);
    return updatedList;
  } catch (error) {
    Logger.log(`Error updating dropdown list: ${error.message}`);
    throw error;
  }
}

function deleteDropdownList(listId) {

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const workflowStepsSheet = ss.getSheetByName(SHEETS.WORKFLOW_STEPS);

    const data = workflowStepsSheet.getDataRange().getValues();
    let rowIndex = -1;
    let workflowStepData = null;

    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === stepId) {
        rowIndex = i + 1;
        workflowStepData = data[i];
        break;
      }
    }

    if (rowIndex === -1) {
      throw new Error('Workflow step not found');
    }

    workflowStepsSheet.deleteRow(rowIndex);

    Logger.log(`Deleted workflow step: ${workflowStepData[2]}`);
    return { message: 'Workflow step deleted successfully' };
  } catch (error) {
    Logger.log(`Error deleting workflow step: ${error.message}`);
    throw error;
  }
}

function recordLogin(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let logSheet = ss.getSheetByName(SHEETS.ADMIN_ACTION_LOGS);

    if (!logSheet) {
      logSheet = initializeAdminActionLogsSheet();
    }

    const logEntry = {
      id: generateId('log'),
      admin_user_id: payload.userId,
      action_type: 'USER_LOGIN',
      target_entity: 'USER',
      target_id: payload.userId,
      details: JSON.stringify({
        email: payload.email,
        ipAddress: payload.ipAddress || 'N/A'
      }),
      timestamp: new Date().toISOString()
    };

    logSheet.appendRow([
      logEntry.id,
      logEntry.admin_user_id,
      logEntry.action_type,
      logEntry.target_entity,
      logEntry.target_id,
      logEntry.details,
      logEntry.timestamp
    ]);

    Logger.log(`Recorded login for user: ${payload.email}`);
    return { message: 'Login recorded successfully' };
  } catch (error) {
    Logger.log(`Error recording login: ${error.message}`);
    throw error;
  }
}

function getSystemHealth() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheets = ss.getSheets();

    const health = {
      status: 'healthy',
      timestamp: new Date().toISOString(),
      spreadsheet_id: SPREADSHEET_ID,
      version: '4.1',
      sheets_count: sheets.length,
      sheets_status: {},
      total_records: 0
    };

    // Check each expected sheet
    Object.values(SHEETS).forEach(sheetName => {
      const sheet = ss.getSheetByName(sheetName);
      if (sheet) {
        const rowCount = Math.max(0, sheet.getLastRow() - 1); // Exclude header
        health.sheets_status[sheetName] = {
          exists: true,
          rows: rowCount
        };
        health.total_records += rowCount;
      } else {
        health.sheets_status[sheetName] = {
          exists: false,
          rows: 0
        };
        health.status = 'degraded';
      }
    });

    return health;
  } catch (error) {
    Logger.log(`Error getting system health: ${error.message}`);
    return {
      status: 'error',
      timestamp: new Date().toISOString(),
      error: error.message
    };
  }
}

/**
 * Converts sheet data to JSON - enhanced with null safety
 */
function getSheetDataAsJSON(sheet) {
  if (!sheet) {
    Logger.log('Warning: getSheetDataAsJSON called with null/undefined sheet');
    return [];
  }

  try {
    const data = sheet.getDataRange().getValues();
    if (data.length < 2) return [];

    const headers = data.shift();
    return data.map(row => {
      let obj = {};
      headers.forEach((header, i) => {
        obj[header] = row[i] !== undefined ? row[i] : null;
      });
      return obj;
    });
  } catch (error) {
    Logger.log(`Error in getSheetDataAsJSON: ${error.message}`);
    return [];
  }
}

/**
 * Creates JSON response - NO HEADER CHAINING
 */
function createJsonResponse(response) {
  try {
    // ✅ CORRECT: Simple approach without chaining
    const output = ContentService.createTextOutput(JSON.stringify(response));
    output.setMimeType(ContentService.MimeType.JSON);
    return output;
  } catch (error) {
    Logger.log('createJsonResponse Error:', error);
    // Fallback response
    const fallback = ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: 'Response creation failed',
      details: error.toString()
    }));
    fallback.setMimeType(ContentService.MimeType.JSON);
    return fallback;
  }
}

/**
 * Generates unique ID - using substring() not substr()
 */
function generateId(prefix) {
  // ✅ AVOID: substr() is deprecated
  // ✅ USE: substring() for future compatibility
  return `${prefix}_${Date.now()}_${Math.random().toString(36).substring(2, 11)}`;
}

/**
 * Generates formatted ticket number with LockService
 */
function generateTicketNumber(companyCode, typeCode) {
  const lock = LockService.getScriptLock();

  try {
    lock.waitLock(30000);

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let countersSheet = ss.getSheetByName(SHEETS.SEQUENCE_COUNTERS);

    if (!countersSheet) {
      countersSheet = initializeSequenceCountersSheet();
    }

    const sequenceName = `${companyCode}-${typeCode}`;
    const year = Utilities.formatDate(new Date(), "Asia/Manila", "yyyy");

    // Get current counter
    const data = countersSheet.getDataRange().getValues();
    let currentNumber = 0;
    let rowIndex = -1;

    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === sequenceName) {
        currentNumber = parseInt(data[i][1], 10) || 0;
        rowIndex = i + 1;
        break;
      }
    }

    const nextNumber = currentNumber + 1;

    // Update or create counter
    if (rowIndex > -1) {
      countersSheet.getRange(rowIndex, 2).setValue(nextNumber);
    } else {
      countersSheet.appendRow([sequenceName, nextNumber]);
    }

    // Format ticket number
    const paddedNumber = String(nextNumber).padStart(8, '0');
    return `${sequenceName}-${year}-${paddedNumber}`;

  } finally {
    lock.releaseLock();
  }
}
