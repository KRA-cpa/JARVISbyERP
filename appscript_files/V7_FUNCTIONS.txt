/**
 * Bulk update step approvers (replace all approvers for a step)
 */
function bulkUpdateStepApprovers(stepId, roleIds, updateInfo = {}) {
  const startTime = Date.now();

  try {
    if (!stepId || !Array.isArray(roleIds)) {
      return createErrorResponse('INVALID_PARAMETERS', 'stepId and roleIds array are required');
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const stepApproversSheet = ss.getSheetByName('step_approvers');

    if (!stepApproversSheet) {
      return createErrorResponse('SHEET_NOT_FOUND', 'step_approvers sheet not found');
    }

    // First, remove all existing approvers for this step
    const data = stepApproversSheet.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1);

    // Collect rows to delete (in reverse order to maintain indices)
    const rowsToDelete = [];
    rows.forEach((row, index) => {
      const rowData = {};
      headers.forEach((header, headerIndex) => {
        rowData[header] = row[headerIndex];
      });

      if (rowData.step_id === stepId) {
        rowsToDelete.unshift(index + 2); // +2 for header row and 0-based index
      }
    });

    // Delete existing approvers
    rowsToDelete.forEach(rowIndex => {
      stepApproversSheet.deleteRow(rowIndex);
    });

    // Add new approvers
    const newApprovers = [];
    const timestamp = new Date().toISOString();

    roleIds.forEach(roleId => {
      if (roleId) { // Skip empty/null role IDs
        const newRow = [
          stepId,                       // step_id
          roleId                        // role_id
        ];

        stepApproversSheet.appendRow(newRow);
        newApprovers.push({
          step_id: stepId,
          role_id: roleId,
          created_at: timestamp,
          created_by: updateInfo.updated_by || 'system'
        });
      }
    });

    const responseTime = Date.now() - startTime;

    return createSuccessResponse(newApprovers,
      `Updated step approvers: removed ${rowsToDelete.length}, added ${newApprovers.length}`,
      { responseTime, stepId, removedCount: rowsToDelete.length, addedCount: newApprovers.length });

  } catch (error) {
    return createErrorResponse('BULK_UPDATE_STEP_APPROVERS_ERROR', error.toString(), {
      stepId, roleIds, updateInfo,
      responseTime: Date.now() - startTime
    });
  }
}

/**
 * Get all workflow steps where a specific role is an approver
 * Useful for role management and impact analysis
 */
function getStepsByRole(roleId, companyId = null) {
  const startTime = Date.now();

  try {
    if (!roleId) {
      return createErrorResponse('MISSING_ROLE_ID', 'Role ID is required');
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const stepApproversSheet = ss.getSheetByName('step_approvers');
    const workflowStepsSheet = ss.getSheetByName('workflow_steps');
    const ticketTypesSheet = ss.getSheetByName('ticket_types');

    if (!stepApproversSheet || !workflowStepsSheet || !ticketTypesSheet) {
      return createErrorResponse('SHEETS_NOT_FOUND', 'Required sheets not found');
    }

    // Get step approvers data
    const stepApproversData = stepApproversSheet.getDataRange().getValues();
    const stepApproversHeaders = stepApproversData[0];
    const stepApproversRows = stepApproversData.slice(1);

    // Get workflow steps data
    const workflowStepsData = workflowStepsSheet.getDataRange().getValues();
    const workflowStepsHeaders = workflowStepsData[0];
    const workflowStepsRows = workflowStepsData.slice(1);

    // Get ticket types data
    const ticketTypesData = ticketTypesSheet.getDataRange().getValues();
    const ticketTypesHeaders = ticketTypesData[0];
    const ticketTypesRows = ticketTypesData.slice(1);

    // Build ticket types lookup
    const ticketTypesLookup = {};
    ticketTypesRows.forEach(row => {
      const ttData = {};
      ticketTypesHeaders.forEach((header, index) => {
        ttData[header] = row[index];
      });
      ticketTypesLookup[ttData.id] = ttData;
    });

    // Build workflow steps lookup
    const workflowStepsLookup = {};
    workflowStepsRows.forEach(row => {
      const stepData = {};
      workflowStepsHeaders.forEach((header, index) => {
        stepData[header] = row[index];
      });
      workflowStepsLookup[stepData.id] = stepData;
    });

    // Find steps where this role is an approver
    const roleSteps = [];
    stepApproversRows.forEach(row => {
      const approverData = {};
      stepApproversHeaders.forEach((header, index) => {
        approverData[header] = row[index];
      });

      if (approverData.role_id === roleId) {
        const stepInfo = workflowStepsLookup[approverData.step_id];
        if (stepInfo && (!companyId || stepInfo.company_id === companyId)) {
          const ticketTypeInfo = ticketTypesLookup[stepInfo.ticket_type_id] || {};

          roleSteps.push({
            step_id: stepInfo.id,
            step_name: stepInfo.name,
            step_type: stepInfo.step_type,

            status_on_reach: stepInfo.status_on_reach,
            is_active: stepInfo.is_active
          });
        }
      }
    });

    // Sort by ticket type, then sort order
    roleSteps.sort((a, b) => {
      if (a.ticket_type_name !== b.ticket_type_name) {
        return a.ticket_type_name.localeCompare(b.ticket_type_name);
      }
      return (a.sort_order || 0) - (b.sort_order || 0);
    });

    const responseTime = Date.now() - startTime;

    return createSuccessResponse(roleSteps,
      `Found ${roleSteps.length} workflow steps where role ${roleId} is an approver`,
      { responseTime, roleId, companyId, stepCount: roleSteps.length });

  } catch (error) {
    return createErrorResponse('GET_STEPS_BY_ROLE_ERROR', error.toString(), {
      roleId, companyId,
      responseTime: Date.now() - startTime
    });
  }
}

/**
 * Clone all approvers from one step to another
 * Useful for workflow template copying
 */
function cloneStepApprovers(fromStepId, toStepId, cloneInfo = {}) {
  const startTime = Date.now();

  try {
    if (!fromStepId || !toStepId) {
      return createErrorResponse('MISSING_STEP_IDS', 'Both fromStepId and toStepId are required');
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const stepApproversSheet = ss.getSheetByName('step_approvers');

    if (!stepApproversSheet) {
      return createErrorResponse('SHEET_NOT_FOUND', 'step_approvers sheet not found');
    }

    // Get source step approvers
    const data = stepApproversSheet.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1);

    const sourceApprovers = [];
    rows.forEach(row => {
      const rowData = {};
      headers.forEach((header, index) => {
        rowData[header] = row[index];
      });

      if (rowData.step_id === fromStepId) {
        sourceApprovers.push(rowData.role_id);
      }
    });

    if (sourceApprovers.length === 0) {
      return createSuccessResponse([],
        `No approvers found in source step ${fromStepId} to clone`,
        { responseTime: Date.now() - startTime, fromStepId, toStepId });
    }

    // Check for existing approvers in target step
    const existingApprovers = [];
    rows.forEach(row => {
      const rowData = {};
      headers.forEach((header, index) => {
        rowData[header] = row[index];
      });

      if (rowData.step_id === toStepId) {
        existingApprovers.push(rowData.role_id);
      }
    });

    // Add only new approvers (avoid duplicates)
    const newApprovers = [];
    const timestamp = new Date().toISOString();
    const clonedBy = cloneInfo.cloned_by || 'system';

    sourceApprovers.forEach(roleId => {
      if (!existingApprovers.includes(roleId)) {
        const newRow = [
          toStepId,                     // step_id
          roleId                        // role_id
        ];

        stepApproversSheet.appendRow(newRow);
        newApprovers.push({
          step_id: toStepId,
          role_id: roleId,
          created_at: timestamp,
          created_by: clonedBy
        });
      }
    });

    const responseTime = Date.now() - startTime;

    return createSuccessResponse(newApprovers,
      `Cloned ${newApprovers.length} approvers from step ${fromStepId} to ${toStepId} (${existingApprovers.length} already existed)`,
      { responseTime, fromStepId, toStepId, clonedCount: newApprovers.length, skippedCount: sourceApprovers.length - newApprovers.length });

  } catch (error) {
    return createErrorResponse('CLONE_STEP_APPROVERS_ERROR', error.toString(), {
      fromStepId, toStepId, cloneInfo,
      responseTime: Date.now() - startTime
    });
  }
}

    const rolesLookup = {};

      });

      companiesLookup[companyData.id] = companyData;

        ttData[header] = row[index];

/**
 * Get all users assigned to a specific role, optionally filtered by company and ticket type
 */
function getUsersByRole(roleId, companyId = null, ticketTypeId = null) {
  const startTime = Date.now();

  try {
    if (!roleId) {
      return createErrorResponse('MISSING_ROLE_ID', 'Role ID is required');
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const userRoleAssignmentsSheet = ss.getSheetByName('user_role_assignments');
    const usersSheet = ss.getSheetByName('users');
    const companiesSheet = ss.getSheetByName('companies');
    const ticketTypesSheet = ss.getSheetByName('ticket_types');

    if (!userRoleAssignmentsSheet || !usersSheet || !companiesSheet || !ticketTypesSheet) {
      return createErrorResponse('SHEETS_NOT_FOUND', 'Required sheets not found');
    }

    // Get all data
    const assignmentsData = userRoleAssignmentsSheet.getDataRange().getValues();
    const assignmentsHeaders = assignmentsData[0];
    const assignmentsRows = assignmentsData.slice(1);

    const usersData = usersSheet.getDataRange().getValues();
    const usersHeaders = usersData[0];
    const usersRows = usersData.slice(1);

    const companiesData = companiesSheet.getDataRange().getValues();
    const companiesHeaders = companiesData[0];
    const companiesRows = companiesData.slice(1);

    const ticketTypesData = ticketTypesSheet.getDataRange().getValues();
    const ticketTypesHeaders = ticketTypesData[0];
    const ticketTypesRows = ticketTypesData.slice(1);

    // Build lookup tables
    const usersLookup = {};
    usersRows.forEach(row => {
      const userData = {};
      usersHeaders.forEach((header, index) => {
        userData[header] = row[index];
      });
      usersLookup[userData.id] = userData;
    });

    const companiesLookup = {};
    companiesRows.forEach(row => {
      const companyData = {};
      companiesHeaders.forEach((header, index) => {
        companyData[header] = row[index];
      });
      companiesLookup[companyData.id] = companyData;
    });

    const ticketTypesLookup = {};
    ticketTypesRows.forEach(row => {
      const ttData = {};
      ticketTypesHeaders.forEach((header, index) => {
        ttData[header] = row[index];
      });
      ticketTypesLookup[ttData.id] = ttData;
    });

    // Filter assignments and get user info
    const roleUsers = [];
    const currentDate = new Date();

    assignmentsRows.forEach(row => {
      const assignmentData = {};
      assignmentsHeaders.forEach((header, index) => {
        assignmentData[header] = row[index];
      });

      // Filter by role ID and optional filters, exclude deactivated assignments
      if (assignmentData.role_id === roleId &&
          (!companyId || assignmentData.company_id === companyId) &&
          (!ticketTypeId || assignmentData.ticket_type_id === ticketTypeId) &&
          !assignmentData.deactivated_at) {

        // Check if assignment is still valid
        const validityEndDate = assignmentData.validity_end_date ? new Date(assignmentData.validity_end_date) : null;
        const isValid = !validityEndDate || validityEndDate > currentDate;

        if (isValid) {
          const userInfo = usersLookup[assignmentData.user_id] || {};
          const companyInfo = companiesLookup[assignmentData.company_id] || {};
          const ticketTypeInfo = ticketTypesLookup[assignmentData.ticket_type_id] || {};

          roleUsers.push({
            user_id: assignmentData.user_id,
            user_name: userInfo.name || 'Unknown User',
            user_email: userInfo.email || 'Unknown Email',
            ticket_type_id: assignmentData.ticket_type_id,
            ticket_type_name: ticketTypeInfo.name || 'Unknown',
            ticket_type_code: ticketTypeInfo.code || 'UNK',
            company_id: assignmentData.company_id,
            company_name: companyInfo.name || 'Unknown Company',
            validity_end_date: assignmentData.validity_end_date,
            created_at: assignmentData.created_at
          });
        }
      }
    });

    // Remove duplicates and sort
    const uniqueUsers = roleUsers.filter((user, index, self) =>
      index === self.findIndex(u => u.user_id === user.user_id && u.ticket_type_id === user.ticket_type_id)
    );

    uniqueUsers.sort((a, b) => {
      if (a.user_name !== b.user_name) {
        return a.user_name.localeCompare(b.user_name);
      }
      return a.ticket_type_name.localeCompare(b.ticket_type_name);
    });

    const responseTime = Date.now() - startTime;

    return createSuccessResponse(uniqueUsers,
      `Found ${uniqueUsers.length} users assigned to role ${roleId}`,
      { responseTime, roleId, companyId, ticketTypeId, userCount: uniqueUsers.length });

  } catch (error) {
    return createErrorResponse('GET_USERS_BY_ROLE_ERROR', error.toString(), {
      roleId, companyId, ticketTypeId,
      responseTime: Date.now() - startTime
    });
  }
}

/**
 * Clone all role assignments from one user to another
 */
function cloneUserRoleAssignments(fromUserId, toUserId, companyId = null, cloneInfo = {}) {
  const startTime = Date.now();

  try {
    if (!fromUserId || !toUserId) {
      return createErrorResponse('MISSING_USER_IDS', 'Both fromUserId and toUserId are required');
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const userRoleAssignmentsSheet = ss.getSheetByName('user_role_assignments');

    if (!userRoleAssignmentsSheet) {
      return createErrorResponse('SHEET_NOT_FOUND', 'user_role_assignments sheet not found');
    }

    // Get source user assignments
    const data = userRoleAssignmentsSheet.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1);

    const sourceAssignments = [];
    const currentDate = new Date();

    rows.forEach(row => {
      const rowData = {};
      headers.forEach((header, index) => {
        rowData[header] = row[index];
      });

      // Get active assignments for source user, optionally filtered by company
      if (rowData.user_id === fromUserId &&
          !rowData.deactivated_at &&
          (!companyId || rowData.company_id === companyId)) {

        // Check validity
        const validityEndDate = rowData.validity_end_date ? new Date(rowData.validity_end_date) : null;
        const isValid = !validityEndDate || validityEndDate > currentDate;

        if (isValid) {
          sourceAssignments.push({
            ticket_type_id: rowData.ticket_type_id,
            role_id: rowData.role_id,
            company_id: rowData.company_id,
            validity_end_date: rowData.validity_end_date
          });
        }
      }
    });

    if (sourceAssignments.length === 0) {
      return createSuccessResponse([],
        `No active role assignments found for user ${fromUserId} to clone`,
        { responseTime: Date.now() - startTime, fromUserId, toUserId });
    }

    // Check existing assignments for target user
    const existingAssignments = new Set();
    rows.forEach(row => {
      const rowData = {};
      headers.forEach((header, index) => {
        rowData[header] = row[index];
      });

      if (rowData.user_id === toUserId && !rowData.deactivated_at) {
        existingAssignments.add(`${rowData.ticket_type_id}-${rowData.role_id}-${rowData.company_id}`);
      }
    });

    // Clone assignments that don't already exist
    const clonedAssignments = [];
    const timestamp = new Date().toISOString();
    const clonedBy = cloneInfo.cloned_by || 'system';

    sourceAssignments.forEach(assignment => {
      const assignmentKey = `${assignment.ticket_type_id}-${assignment.role_id}-${assignment.company_id}`;

      if (!existingAssignments.has(assignmentKey)) {
        const newRow = [
          toUserId,                           // user_id
          assignment.ticket_type_id,          // ticket_type_id
          assignment.role_id,                 // role_id
          assignment.validity_end_date || '', // validity_end_date
          assignment.company_id,              // company_id
          timestamp,                          // created_at
          clonedBy,                           // created_by
          '',                                 // updated_at
          '',                                 // updated_by
          '',                                 // deactivated_at
          '',                                 // deactivated_by
          ''                                  // deactivation_reason
        ];

        userRoleAssignmentsSheet.appendRow(newRow);
        clonedAssignments.push({
          user_id: toUserId,
          ticket_type_id: assignment.ticket_type_id,
          role_id: assignment.role_id,
          company_id: assignment.company_id,
          validity_end_date: assignment.validity_end_date,
          created_at: timestamp,
          created_by: clonedBy
        });
      }
    });

    const responseTime = Date.now() - startTime;

    return createSuccessResponse(clonedAssignments,
      `Cloned ${clonedAssignments.length} role assignments from user ${fromUserId} to ${toUserId} (${sourceAssignments.length - clonedAssignments.length} already existed)`,
      { responseTime, fromUserId, toUserId, companyId, clonedCount: clonedAssignments.length, skippedCount: sourceAssignments.length - clonedAssignments.length });

  } catch (error) {
    return createErrorResponse('CLONE_USER_ROLE_ASSIGNMENTS_ERROR', error.toString(), {
      fromUserId, toUserId, companyId, cloneInfo,
      responseTime: Date.now() - startTime
    });
  }
}

/**
 * Create SLA configuration for a workflow step
 */
function createStepSLA(payload) {
  const startTime = Date.now();

  try {
    const { step_id, duration, unit, exclude_weekends, created_by } = payload;

    if (!step_id || !duration || !unit) {
      return createErrorResponse('MISSING_REQUIRED_FIELDS', 'step_id, duration, and unit are required');
    }

    // Validate unit
    const validUnits = ['hours', 'days'];
    if (!validUnits.includes(unit)) {
      return createErrorResponse('INVALID_UNIT', 'Unit must be either "hours" or "days"');
    }

    // Validate duration
    if (typeof duration !== 'number' || duration <= 0) {
      return createErrorResponse('INVALID_DURATION', 'Duration must be a positive number');
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const stepSLAsSheet = ss.getSheetByName('step_slas');

    if (!stepSLAsSheet) {
      return createErrorResponse('SHEET_NOT_FOUND', 'step_slas sheet not found');
    }

    // Check if SLA already exists for this step
    const data = stepSLAsSheet.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1);

    const existingSLA = rows.find(row => {
      const rowData = {};
      headers.forEach((header, index) => {
        rowData[header] = row[index];
      });
      return rowData.step_id === step_id;
    });

    if (existingSLA) {
      return createErrorResponse('SLA_EXISTS', 'SLA configuration already exists for this step');
    }

    // Create new SLA
    const newRow = [
      step_id,                          // step_id
      duration,                         // duration
      unit,                             // unit
      exclude_weekends === true,        // exclude_weekends
      new Date().toISOString(),         // created_at
      created_by || 'system',           // created_by
      '',                               // updated_at
      ''                                // updated_by
    ];

    stepSLAsSheet.appendRow(newRow);

    const newSLA = {
      step_id,
      duration,
      unit,
      exclude_weekends: exclude_weekends === true,
      created_at: newRow[4],
      created_by: newRow[5]
    };

    const responseTime = Date.now() - startTime;

    return createSuccessResponse(newSLA,
      'Step SLA configuration created successfully',
      { responseTime, step_id, duration, unit });

  } catch (error) {
    return createErrorResponse('CREATE_STEP_SLA_ERROR', error.toString(), {
      payload,
      responseTime: Date.now() - startTime
    });
  }
}

/**
 * Calculate SLA due date for a ticket at a specific step
 * Considers Philippine timezone (UTC+8) and weekend exclusions
 */
function calculateSLADueDate(stepId, startDate = null) {
  const startTime = Date.now();

  try {
    if (!stepId) {
      return createErrorResponse('MISSING_STEP_ID', 'Step ID is required');
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const stepSLAsSheet = ss.getSheetByName('step_slas');

    if (!stepSLAsSheet) {
      return createErrorResponse('SHEET_NOT_FOUND', 'step_slas sheet not found');
    }

    // Get SLA configuration for the step
    const data = stepSLAsSheet.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1);

    let slaConfig = null;
    rows.forEach(row => {
      const rowData = {};
      headers.forEach((header, index) => {
        rowData[header] = row[index];
      });

      if (rowData.step_id === stepId) {
        slaConfig = rowData;
      }
    });

    if (!slaConfig) {
      return createErrorResponse('SLA_NOT_FOUND', 'No SLA configuration found for this step');
    }

    // Start date (default to current Philippine time)
    const philippineOffset = 8; // UTC+8
    const now = new Date();
    const philippineNow = new Date(now.getTime() + (philippineOffset * 60 * 60 * 1000));
    const calculationStartDate = startDate ? new Date(startDate) : philippineNow;

    // Calculate due date
    let dueDate = new Date(calculationStartDate);
    const duration = slaConfig.duration;
    const unit = slaConfig.unit;
    const excludeWeekends = slaConfig.exclude_weekends;

    if (unit === 'hours') {
      if (excludeWeekends) {
        // Add business hours only (skip weekends)
        let hoursToAdd = duration;
        let currentDate = new Date(dueDate);

        while (hoursToAdd > 0) {
          const dayOfWeek = currentDate.getDay(); // 0=Sunday, 6=Saturday

          if (dayOfWeek !== 0 && dayOfWeek !== 6) { // Not weekend
            const hoursToEndOfDay = 24 - currentDate.getHours();
            if (hoursToAdd <= hoursToEndOfDay) {
              currentDate.setHours(currentDate.getHours() + hoursToAdd);
              hoursToAdd = 0;
            } else {
              hoursToAdd -= hoursToEndOfDay;
              // Move to next day
              currentDate.setDate(currentDate.getDate() + 1);
              currentDate.setHours(0, 0, 0, 0);
            }
          } else {
            // Skip weekend day
            currentDate.setDate(currentDate.getDate() + 1);
            currentDate.setHours(0, 0, 0, 0);
          }
        }
        dueDate = currentDate;
      } else {
        // Simple hour addition
        dueDate.setHours(dueDate.getHours() + duration);
      }
    } else if (unit === 'days') {
      if (excludeWeekends) {
        // Add business days only
        let daysToAdd = duration;
        let currentDate = new Date(dueDate);

        while (daysToAdd > 0) {
          currentDate.setDate(currentDate.getDate() + 1);
          const dayOfWeek = currentDate.getDay();

          if (dayOfWeek !== 0 && dayOfWeek !== 6) { // Not weekend
            daysToAdd--;
          }
        }
        dueDate = currentDate;
      } else {
        // Simple day addition
        dueDate.setDate(dueDate.getDate() + duration);
      }
    }

    // Calculate time remaining
    const timeRemaining = dueDate.getTime() - philippineNow.getTime();
    const isOverdue = timeRemaining < 0;

    // Format time remaining
    const absTimeRemaining = Math.abs(timeRemaining);
    const daysRemaining = Math.floor(absTimeRemaining / (24 * 60 * 60 * 1000));
    const hoursRemaining = Math.floor((absTimeRemaining % (24 * 60 * 60 * 1000)) / (60 * 60 * 1000));
    const minutesRemaining = Math.floor((absTimeRemaining % (60 * 60 * 1000)) / (60 * 1000));

    let timeRemainingText = '';
    if (daysRemaining > 0) {
      timeRemainingText = `${daysRemaining}d ${hoursRemaining}h ${minutesRemaining}m`;
    } else if (hoursRemaining > 0) {
      timeRemainingText = `${hoursRemaining}h ${minutesRemaining}m`;
    } else {
      timeRemainingText = `${minutesRemaining}m`;
    }

    const slaStatus = isOverdue ? 'OVERDUE' : (timeRemaining < 60 * 60 * 1000 ? 'DUE_SOON' : 'ON_TIME'); // Due soon if less than 1 hour

    const responseTime = Date.now() - startTime;

    return createSuccessResponse({
      step_id: stepId,
      start_date: calculationStartDate.toISOString(),
      due_date: dueDate.toISOString(),
      duration: duration,
      unit: unit,
      exclude_weekends: excludeWeekends,
      time_remaining_ms: timeRemaining,
      time_remaining_text: (isOverdue ? 'OVERDUE by ' : '') + timeRemainingText,
      sla_status: slaStatus,
      is_overdue: isOverdue,
      philippine_time: philippineNow.toISOString()
    }, `SLA due date calculated for step ${stepId}`,
    { responseTime, stepId, slaStatus });

  } catch (error) {
    return createErrorResponse('CALCULATE_SLA_DUE_DATE_ERROR', error.toString(), {
      stepId, startDate,
      responseTime: Date.now() - startTime
    });
  }
}

/**
 * Clone SLA configurations from one ticket type to another
 */
function cloneStepSLAs(fromTicketTypeId, fromCompanyId, toTicketTypeId, toCompanyId, cloneInfo = {}) {
  const startTime = Date.now();

  try {
    if (!fromTicketTypeId || !fromCompanyId || !toTicketTypeId || !toCompanyId) {
      return createErrorResponse('MISSING_REQUIRED_FIELDS', 'All ticket type and company IDs are required');
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const stepSLAsSheet = ss.getSheetByName('step_slas');
    const workflowStepsSheet = ss.getSheetByName('workflow_steps');

    if (!stepSLAsSheet || !workflowStepsSheet) {
      return createErrorResponse('SHEETS_NOT_FOUND', 'Required sheets not found: step_slas or workflow_steps');
    }

    // Get workflow steps for source and target
    const stepsData = workflowStepsSheet.getDataRange().getValues();
    const stepsHeaders = stepsData[0];
    const stepsRows = stepsData.slice(1);

    // Build step mappings
    const sourceSteps = [];
    const targetSteps = [];

    stepsRows.forEach(row => {
      const stepData = {};
      stepsHeaders.forEach((header, index) => {
        stepData[header] = row[index];
      });

      if (stepData.ticket_type_id === fromTicketTypeId && stepData.company_id === fromCompanyId) {
        sourceSteps.push(stepData);
      } else if (stepData.ticket_type_id === toTicketTypeId && stepData.company_id === toCompanyId) {
        targetSteps.push(stepData);
      }
    });

    // Sort by sort_order for mapping
    sourceSteps.sort((a, b) => (a.sort_order || 0) - (b.sort_order || 0));
    targetSteps.sort((a, b) => (a.sort_order || 0) - (b.sort_order || 0));

    if (sourceSteps.length === 0) {
      return createSuccessResponse([],
        `No workflow steps found for source ticket type ${fromTicketTypeId}`,
        { responseTime: Date.now() - startTime });
    }

    if (targetSteps.length === 0) {
      return createSuccessResponse([],
        `No workflow steps found for target ticket type ${toTicketTypeId}`,
        { responseTime: Date.now() - startTime });
    }

    // Get existing SLA configurations
    const slaData = stepSLAsSheet.getDataRange().getValues();
    const slaHeaders = slaData[0];
    const slaRows = slaData.slice(1);

    const sourceSLAs = {};
    slaRows.forEach(row => {
      const slaRowData = {};
      slaHeaders.forEach((header, index) => {
        slaRowData[header] = row[index];
      });

      sourceSLAs[slaRowData.step_id] = slaRowData;
    });

    // Clone SLA configurations
    const clonedSLAs = [];
    const timestamp = new Date().toISOString();
    const clonedBy = cloneInfo.cloned_by || 'system';

    // Map source steps to target steps by position
    const maxMappings = Math.min(sourceSteps.length, targetSteps.length);

    for (let i = 0; i < maxMappings; i++) {
      const sourceStep = sourceSteps[i];
      const targetStep = targetSteps[i];
      const sourceSLA = sourceSLAs[sourceStep.id];

      if (sourceSLA) {
        // Check if target step already has SLA
        const existingTargetSLA = slaRows.find(row => {
          const rowData = {};
          slaHeaders.forEach((header, index) => {
            rowData[header] = row[index];
          });
          return rowData.step_id === targetStep.id;
        });

        if (!existingTargetSLA) {
          const newRow = [
            targetStep.id,                    // step_id
            sourceSLA.duration,               // duration
            sourceSLA.unit,                   // unit
            sourceSLA.exclude_weekends,       // exclude_weekends
            timestamp,                        // created_at
            clonedBy,                         // created_by
            '',                               // updated_at
            ''                                // updated_by
          ];

          stepSLAsSheet.appendRow(newRow);
          clonedSLAs.push({
            step_id: targetStep.id,
            step_name: targetStep.name,
            duration: sourceSLA.duration,
            unit: sourceSLA.unit,
            exclude_weekends: sourceSLA.exclude_weekends,
            created_at: timestamp,
            created_by: clonedBy
          });
        }
      }
    }

    const responseTime = Date.now() - startTime;

    return createSuccessResponse(clonedSLAs,
      `Cloned ${clonedSLAs.length} SLA configurations from ${fromTicketTypeId} to ${toTicketTypeId}`,
      { responseTime, fromTicketTypeId, fromCompanyId, toTicketTypeId, toCompanyId, clonedCount: clonedSLAs.length });

  } catch (error) {
    return createErrorResponse('CLONE_STEP_SLAS_ERROR', error.toString(), {
      fromTicketTypeId, fromCompanyId, toTicketTypeId, toCompanyId, cloneInfo,
      responseTime: Date.now() - startTime
    });
  }
}
