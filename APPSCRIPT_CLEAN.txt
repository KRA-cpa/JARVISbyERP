function extractPayload(data, requiredFields) {
  // Handle both { action, payload } and { action, ...fields } formats
  let payload = data.payload;

  if (!payload) {
    // Extract from root level (spread format)
    payload = {};
    requiredFields.forEach(field => {
      if (data[field] !== undefined) {
        payload[field] = data[field];
      }
    });
  }

  // ✅ PROPER VALIDATION: Check payload exists before accessing properties
  if (!payload) {
    throw new Error('Payload is required');
  }

  return payload;
}

function updateCompany(payload) {

    // Update the company
    const updatedCompany = {
      id: payload.id,
      name: payload.name.trim(),

function createTicket(payload) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const ticketsSheet = ss.getSheetByName(SHEETS.TICKETS);

    const now = new Date();
    const newTicketId = generateId('ticket');

    // Generate ticket number if company and type info available
    let ticketNumber = newTicketId; // Fallback
    if (payload.company_id && payload.ticket_type_id) {
      try {
        const companies = getSheetDataAsJSON(ss.getSheetByName(SHEETS.COMPANIES));
        const ticketTypes = getSheetDataAsJSON(ss.getSheetByName(SHEETS.TICKET_TYPES));

        const company = companies.find(c => c.id === payload.company_id);
        const ticketType = ticketTypes.find(t => t.id === payload.ticket_type_id);

        if (company && ticketType) {
          ticketNumber = generateTicketNumber(company.code, ticketType.code);
        }
      } catch (error) {
        Logger.log('Could not generate formatted ticket number:', error.message);
      }
    }

    // Create ticket entry
    const newTicket = {
      id: newTicketId,
      ticket_number: ticketNumber,
      title: payload.title,
      ticket_type_id: payload.ticket_type_id || '',
      requester_id: payload.requester_id || '',
      status: 'New',
      current_step_id: payload.current_step_id || '',
      step_due_date: payload.step_due_date || '',
      created_at: now.toISOString(),
      updated_at: now.toISOString(),
      company_id: payload.company_id || ''
    };

    // Add to sheet
    ticketsSheet.appendRow([
      newTicket.id,
      newTicket.ticket_number,
      newTicket.title,
      newTicket.ticket_type_id,
      newTicket.requester_id,
      newTicket.status,
      newTicket.current_step_id,
      newTicket.step_due_date,
      newTicket.created_at,
      newTicket.updated_at,
      newTicket.company_id
    ]);

    Logger.log(`Created ticket: ${newTicket.ticket_number}`);
    return newTicket;
  } catch (error) {
    Logger.log(`Error creating ticket: ${error.message}`);
    throw error;
  }
}

      company_id: payload.company_id === 'global' ? null : payload.company_id,
      created_at: data[rowIndex - 1][3],
      updated_at: new Date().toISOString()
    };

    rolesSheet.getRange(rowIndex, 1, 1, 5).setValues([[
      updatedRole.id,
      updatedRole.name,
      updatedRole.company_id || '',
      updatedRole.created_at,
      updatedRole.updated_at
    ]]);

    Logger.log(`Updated role: ${updatedRole.name}`);
    return updatedRole;
  } catch (error) {
    Logger.log(`Error updating role: ${error.message}`);
    throw error;
  }
}

function deleteRole(roleId) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const rolesSheet = ss.getSheetByName(SHEETS.ROLES);

    const data = rolesSheet.getDataRange().getValues();
    let rowIndex = -1;
    let roleData = null;

    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === roleId) {
        rowIndex = i + 1;
        roleData = data[i];
        break;
      }
    }

    if (rowIndex === -1) {
      throw new Error('Role not found');
    }

    rolesSheet.deleteRow(rowIndex);

    Logger.log(`Deleted role: ${roleData[1]}`);
    return { message: 'Role deleted successfully' };
  } catch (error) {
    Logger.log(`Error deleting role: ${error.message}`);
    throw error;
  }
}

/**
 * Converts sheet data to JSON - safe implementation
 */
function getSheetDataAsJSON(sheet) {
  if (!sheet) {
    Logger.log('Warning: getSheetDataAsJSON called with null/undefined sheet');
    return [];
  }

  try {
    const data = sheet.getDataRange().getValues();
    if (data.length < 2) return [];

    const headers = data.shift();
    return data.map(row => {
      let obj = {};
      headers.forEach((header, i) => {
        obj[header] = row[i] !== undefined ? row[i] : null;
      });
      return obj;
    });
  } catch (error) {
    Logger.log(`Error in getSheetDataAsJSON: ${error.message}`);
    return [];
  }
}

/**
 * Creates JSON response - NO HEADER CHAINING
 */
function createJsonResponse(response) {
  try {
    // ✅ CORRECT: Simple approach without chaining
    const output = ContentService.createTextOutput(JSON.stringify(response));
    output.setMimeType(ContentService.MimeType.JSON);
    return output;
  } catch (error) {
    Logger.log('createJsonResponse Error:', error);
    // Fallback response
    const fallback = ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: 'Response creation failed',
      details: error.toString()
    }));
    fallback.setMimeType(ContentService.MimeType.JSON);
    return fallback;
  }
}

/**
 * Generates unique ID - using substring() not substr()
 */
function generateId(prefix) {
  // ✅ AVOID: substr() is deprecated
  // ✅ USE: substring() for future compatibility
  return `${prefix}_${Date.now()}_${Math.random().toString(36).substring(2, 11)}`;
}

/**
 * Generates formatted ticket number with LockService
 */
function generateTicketNumber(companyCode, typeCode) {
  const lock = LockService.getScriptLock();

  try {
    lock.waitLock(30000);

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let countersSheet = ss.getSheetByName(SHEETS.SEQUENCE_COUNTERS);

    if (!countersSheet) {
      countersSheet = initializeSequenceCountersSheet();
    }

    const sequenceName = `${companyCode}-${typeCode}`;
    const year = Utilities.formatDate(new Date(), "Asia/Manila", "yyyy");

    // Get current counter
    const data = countersSheet.getDataRange().getValues();
    let currentNumber = 0;
    let rowIndex = -1;

    for (let i = 1; i < data.length; i++) {
      if (data[i][0] === sequenceName) {
        currentNumber = parseInt(data[i][1], 10) || 0;
        rowIndex = i + 1;
        break;
      }
    }

    const nextNumber = currentNumber + 1;

    // Update or create counter
    if (rowIndex > -1) {
      countersSheet.getRange(rowIndex, 2).setValue(nextNumber);
    } else {
      countersSheet.appendRow([sequenceName, nextNumber]);
    }

    // Format ticket number
    const paddedNumber = String(nextNumber).padStart(8, '0');
    return `${sequenceName}-${year}-${paddedNumber}`;

  } finally {
    lock.releaseLock();
  }
}
