/**
 * Get all ticket tags for a company (includes global tags)
 * @param {string} companyId - Company ID to filter tags
 * @param {boolean} includeGlobal - Whether to include global tags (default: true)
 * @returns {Object} API response with tags data
 */
function getTicketTags(companyId = null, includeGlobal = true) {
    try {
        const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
        const sheet = spreadsheet.getSheetByName(TAG_SHEETS.TICKET_TAGS);

        if (!sheet) {
            return createErrorResponse("Ticket tags sheet not found");
        }

        const data = sheet.getDataRange().getValues();
        const headers = data[0];
        const rows = data.slice(1);

        let tags = rows
            .filter(row => row[headers.indexOf('is_active')] === true)
            .map(row => ({
                id: row[headers.indexOf('id')],
                name: row[headers.indexOf('name')],
                color: row[headers.indexOf('color')],
                description: row[headers.indexOf('description')],
                tag_category: row[headers.indexOf('tag_category')],
                parent_tag_id: row[headers.indexOf('parent_tag_id')],
                company_id: row[headers.indexOf('company_id')],
                is_global: row[headers.indexOf('is_global')],
                usage_count: row[headers.indexOf('usage_count')] || 0,
                created_by_user_id: row[headers.indexOf('created_by_user_id')],
                created_at: row[headers.indexOf('created_at')],
                updated_at: row[headers.indexOf('updated_at')]
            }));

        // Filter by company and global settings
        if (companyId) {
            tags = tags.filter(tag =>
                tag.company_id === companyId ||
                (includeGlobal && tag.is_global === true)
            );
        }

        // Sort by usage count (most used first) then by name
        tags.sort((a, b) => {
            if (b.usage_count !== a.usage_count) {
                return b.usage_count - a.usage_count;
            }
            return a.name.localeCompare(b.name);
        });

        return createSuccessResponse(tags);

    } catch (error) {
        return createErrorResponse("Failed to get ticket tags: " + error.message);
    }
}

/**

                sheet.getRange(rowIndex + 1, index + 1).setValue(updateFields[header]);

                updatedTag[header] = data[rowIndex][index];
            }

        // Log admin action
        logAdminAction(updateData.updated_by_user_id, 'UPDATE_TAG', 'ticket_tag', tagId, updateData);

        return createSuccessResponse(updatedTag);

/**
 * Remove tags from a ticket
 * @param {string} ticketId - Ticket ID
 * @param {Array} tagIds - Array of tag IDs to remove
 * @param {string} removedByUserId - User removing the tags
 * @param {string} reason - Reason for removal
 * @returns {Object} API response
 */
function removeTagsFromTicket(ticketId, tagIds, removedByUserId, reason = 'user_request') {
    try {
        const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
        const assignmentSheet = spreadsheet.getSheetByName(TAG_SHEETS.TICKET_TAG_ASSIGNMENTS);

        if (!assignmentSheet) {
            return createErrorResponse("Tag assignments sheet not found");
        }

        const data = assignmentSheet.getDataRange().getValues();
        const headers = data[0];
        const removedCount = 0;

        // Find and deactivate assignments
        for (let i = 1; i < data.length; i++) {
            const row = data[i];
            const ticketIdCol = row[headers.indexOf('ticket_id')];
            const tagIdCol = row[headers.indexOf('tag_id')];
            const isActiveCol = row[headers.indexOf('is_active')];

            if (ticketIdCol === ticketId && tagIds.includes(tagIdCol) && isActiveCol === true) {
                // Deactivate this assignment
                const deletionFields = deleteRecord(removedByUserId, reason);
                headers.forEach((header, index) => {
                    if (deletionFields.hasOwnProperty(header)) {
                        assignmentSheet.getRange(i + 1, index + 1).setValue(deletionFields[header]);
                    }
                });
                removedCount++;
            }
        }

        // Update ticket tag count
        updateTicketTagCount(ticketId);

        // Log action
        logTicketAction(ticketId, removedByUserId, 'TAGS_REMOVED', {
            tag_ids: tagIds,
            removal_reason: reason
        });

        return createSuccessResponse({
            message: `${removedCount} tags removed successfully`
        });

    } catch (error) {
        return createErrorResponse("Failed to remove tags: " + error.message);
    }
}

/**
 * Get tags assigned to a specific ticket
 * @param {string} ticketId - Ticket ID
 * @returns {Object} API response with assigned tags
 */
function getTicketTags(ticketId) {
    try {
        const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
        const assignmentSheet = spreadsheet.getSheetByName(TAG_SHEETS.TICKET_TAG_ASSIGNMENTS);
        const tagSheet = spreadsheet.getSheetByName(TAG_SHEETS.TICKET_TAGS);

        // Get active assignments for this ticket
        const assignmentData = assignmentSheet.getDataRange().getValues();

        const activeAssignments = assignmentData.slice(1)

                row[assignmentHeaders.indexOf('ticket_id')] === ticketId &&

            const tagRow = tagData.find(row =>
                row[tagHeaders.indexOf('id')] === assignment.tag_id &&

        return createErrorResponse("Failed to get ticket tags: " + error.message);

/**
 * Search tickets by tags
 * @param {Array} tagIds - Array of tag IDs to search for
 * @param {string} operator - Search operator ('AND' or 'OR')
 * @param {string} companyId - Company ID filter (optional)
 * @returns {Object} API response with matching tickets
 */
function searchTicketsByTags(tagIds, operator = 'OR', companyId = null) {
    try {
        const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
        const assignmentSheet = spreadsheet.getSheetByName(TAG_SHEETS.TICKET_TAG_ASSIGNMENTS);
        const ticketSheet = spreadsheet.getSheetByName(SHEETS.TICKETS);

        if (!assignmentSheet || !ticketSheet) {
            return createErrorResponse("Required sheets not found");
        }

        // Get active tag assignments
        const assignmentData = assignmentSheet.getDataRange().getValues();
        const assignmentHeaders = assignmentData[0];

        const activeAssignments = assignmentData.slice(1)
            .filter(row =>
                tagIds.includes(row[assignmentHeaders.indexOf('tag_id')]) &&
                row[assignmentHeaders.indexOf('is_active')] === true
            );

        // Group assignments by ticket
        const ticketTagMap = {};
        activeAssignments.forEach(row => {
            const ticketId = row[assignmentHeaders.indexOf('ticket_id')];
            const tagId = row[assignmentHeaders.indexOf('tag_id')];

            if (!ticketTagMap[ticketId]) {
                ticketTagMap[ticketId] = [];
            }
            ticketTagMap[ticketId].push(tagId);
        });

        // Filter tickets based on operator
        const matchingTicketIds = Object.keys(ticketTagMap).filter(ticketId => {
            const ticketTags = ticketTagMap[ticketId];

            if (operator === 'AND') {
                return tagIds.every(tagId => ticketTags.includes(tagId));
            } else {
                return tagIds.some(tagId => ticketTags.includes(tagId));
            }
        });

        // Get ticket details
        const ticketData = ticketSheet.getDataRange().getValues();
        const ticketHeaders = ticketData[0];

        let tickets = ticketData.slice(1)
            .filter(row =>
                matchingTicketIds.includes(row[ticketHeaders.indexOf('id')]) &&
                row[ticketHeaders.indexOf('is_active')] === true &&
                (!companyId || row[ticketHeaders.indexOf('company_id')] === companyId)
            )
            .map(row => ({
                id: row[ticketHeaders.indexOf('id')],
                ticket_number: row[ticketHeaders.indexOf('ticket_number')],
                title: row[ticketHeaders.indexOf('title')],
                ticket_type_id: row[ticketHeaders.indexOf('ticket_type_id')],
                requester_id: row[ticketHeaders.indexOf('requester_id')],
                status: row[ticketHeaders.indexOf('status')],
                company_id: row[ticketHeaders.indexOf('company_id')],
                created_at: row[ticketHeaders.indexOf('created_at')],
                updated_at: row[ticketHeaders.indexOf('updated_at')],
                matched_tags: ticketTagMap[row[ticketHeaders.indexOf('id')]]
            }));

        return createSuccessResponse({
            tickets: tickets,
            search_criteria: {
                tag_ids: tagIds,
                operator: operator,
                company_id: companyId
            },
            total_matches: tickets.length
        });

    } catch (error) {
        return createErrorResponse("Failed to search tickets by tags: " + error.message);
    }
}

/**
 * Approve or reject a collaboration request
 * @param {string} requestId - Request ID
 * @param {Object} responseData - Approval/rejection details
 * @returns {Object} API response
 */
function respondToCollaborationRequest(requestId, responseData) {
    try {
        const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
        const requestSheet = spreadsheet.getSheetByName(TAG_SHEETS.COLLABORATION_REQUESTS);

        if (!requestSheet) {
            return createErrorResponse("Collaboration requests sheet not found");
        }

        const data = requestSheet.getDataRange().getValues();
        const headers = data[0];
        const idColumnIndex = headers.indexOf('id');

        // Find the request row
        const rowIndex = data.findIndex(row => row[idColumnIndex] === requestId);
        if (rowIndex === -1) {
            return createErrorResponse("Request not found");
        }

        const request = data[rowIndex];
        const currentTimestamp = getCurrentTimestamp();

        if (responseData.action === 'approve') {
            // Update request as approved
            requestSheet.getRange(rowIndex + 1, headers.indexOf('request_status') + 1).setValue('approved');
            requestSheet.getRange(rowIndex + 1, headers.indexOf('approved_by_user_id') + 1).setValue(responseData.responded_by_user_id);
            requestSheet.getRange(rowIndex + 1, headers.indexOf('approved_at') + 1).setValue(currentTimestamp);

            // Create collaboration
            const shareData = {
                shared_by_user_id: responseData.responded_by_user_id,
                shared_with_user_id: request[headers.indexOf('requested_by_user_id')],
                permission_level: request[headers.indexOf('permission_level')],
                share_reason: 'Approved collaboration request'
            };

            const shareResult = shareTicket(request[headers.indexOf('ticket_id')], shareData);

            // Create notification
            createCollaborationNotification(requestId, 'REQUEST_APPROVED', request[headers.indexOf('requested_by_user_id')],
                'Your collaboration request has been approved');

            return shareResult;

        } else if (responseData.action === 'reject') {
            // Update request as rejected
            requestSheet.getRange(rowIndex + 1, headers.indexOf('request_status') + 1).setValue('rejected');
            requestSheet.getRange(rowIndex + 1, headers.indexOf('rejected_by_user_id') + 1).setValue(responseData.responded_by_user_id);
            requestSheet.getRange(rowIndex + 1, headers.indexOf('rejected_at') + 1).setValue(currentTimestamp);
            requestSheet.getRange(rowIndex + 1, headers.indexOf('rejection_reason') + 1).setValue(responseData.rejection_reason || '');

            // Create notification
            createCollaborationNotification(requestId, 'REQUEST_REJECTED', request[headers.indexOf('requested_by_user_id')],
                `Your collaboration request has been rejected: ${responseData.rejection_reason || 'No reason provided'}`);

            return createSuccessResponse({ message: 'Request rejected successfully' });
        }

        return createErrorResponse("Invalid action specified");

    } catch (error) {
        return createErrorResponse("Failed to respond to collaboration request: " + error.message);
    }
}

/**
 * Get shared tickets for a user
 * @param {string} userId - User ID
 * @param {string} accessType - Type of access ('shared_by_me', 'shared_with_me', 'all')
 * @returns {Object} API response with shared tickets
 */
function getSharedTickets(userId, accessType = 'all') {
    try {
        const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
        const collaborationSheet = spreadsheet.getSheetByName(TAG_SHEETS.TICKET_COLLABORATIONS);
        const ticketSheet = spreadsheet.getSheetByName(SHEETS.TICKETS);

        if (!collaborationSheet || !ticketSheet) {
            return createErrorResponse("Required sheets not found");
        }

        const collaborationData = collaborationSheet.getDataRange().getValues();
        const collaborationHeaders = collaborationData[0];

        // Filter collaborations based on access type
        let relevantCollaborations = collaborationData.slice(1)
            .filter(row => row[collaborationHeaders.indexOf('is_active')] === true);

        if (accessType === 'shared_by_me') {
            relevantCollaborations = relevantCollaborations.filter(row =>
                row[collaborationHeaders.indexOf('shared_by_user_id')] === userId
            );
        } else if (accessType === 'shared_with_me') {
            relevantCollaborations = relevantCollaborations.filter(row =>
                row[collaborationHeaders.indexOf('shared_with_user_id')] === userId
            );
        } else {
            relevantCollaborations = relevantCollaborations.filter(row =>
                row[collaborationHeaders.indexOf('shared_by_user_id')] === userId ||
                row[collaborationHeaders.indexOf('shared_with_user_id')] === userId
            );
        }

        // Get ticket details
        const ticketData = ticketSheet.getDataRange().getValues();
        const ticketHeaders = ticketData[0];

        const sharedTickets = relevantCollaborations.map(collaboration => {
            const ticketId = collaboration[collaborationHeaders.indexOf('ticket_id')];
            const ticketRow = ticketData.find(row =>
                row[ticketHeaders.indexOf('id')] === ticketId &&
                row[ticketHeaders.indexOf('is_active')] === true
            );

            if (ticketRow) {
                return {
                    collaboration_id: collaboration[collaborationHeaders.indexOf('id')],
                    ticket_id: ticketId,
                    ticket_number: ticketRow[ticketHeaders.indexOf('ticket_number')],
                    title: ticketRow[ticketHeaders.indexOf('title')],
                    status: ticketRow[ticketHeaders.indexOf('status')],
                    shared_by_user_id: collaboration[collaborationHeaders.indexOf('shared_by_user_id')],
                    shared_with_user_id: collaboration[collaborationHeaders.indexOf('shared_with_user_id')],
                    permission_level: collaboration[collaborationHeaders.indexOf('permission_level')],
                    can_view: collaboration[collaborationHeaders.indexOf('can_view')],
                    can_comment: collaboration[collaborationHeaders.indexOf('can_comment')],
                    can_edit: collaboration[collaborationHeaders.indexOf('can_edit')],
                    can_approve: collaboration[collaborationHeaders.indexOf('can_approve')],
                    expiration_date: collaboration[collaborationHeaders.indexOf('expiration_date')],
                    shared_at: collaboration[collaborationHeaders.indexOf('created_at')]
                };
            }
            return null;
        }).filter(ticket => ticket !== null);

        return createSuccessResponse({
            shared_tickets: sharedTickets,
            access_type: accessType,
            total_count: sharedTickets.length
        });

    } catch (error) {
        return createErrorResponse("Failed to get shared tickets: " + error.message);
    }
}

/**
 * Revoke collaboration access
 * @param {string} collaborationId - Collaboration ID to revoke
 * @param {string} revokedByUserId - User revoking access
 * @param {string} reason - Reason for revocation
 * @returns {Object} API response
 */
function revokeCollaboration(collaborationId, revokedByUserId, reason = 'admin_request') {
    try {
        const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
        const collaborationSheet = spreadsheet.getSheetByName(TAG_SHEETS.TICKET_COLLABORATIONS);

        if (!collaborationSheet) {
            return createErrorResponse("Collaborations sheet not found");
        }

        const data = collaborationSheet.getDataRange().getValues();
        const headers = data[0];
        const idColumnIndex = headers.indexOf('id');

        // Find the collaboration row
        const rowIndex = data.findIndex(row => row[idColumnIndex] === collaborationId);
        if (rowIndex === -1) {
            return createErrorResponse("Collaboration not found");
        }

        const collaboration = data[rowIndex];
        const currentTimestamp = getCurrentTimestamp();

        // Revoke access
        collaborationSheet.getRange(rowIndex + 1, headers.indexOf('is_active') + 1).setValue(false);
        collaborationSheet.getRange(rowIndex + 1, headers.indexOf('revoked_at') + 1).setValue(currentTimestamp);
        collaborationSheet.getRange(rowIndex + 1, headers.indexOf('revoked_by') + 1).setValue(revokedByUserId);
        collaborationSheet.getRange(rowIndex + 1, headers.indexOf('revocation_reason') + 1).setValue(reason);

        // Update ticket collaboration status
        updateTicketCollaborationStatus(collaboration[headers.indexOf('ticket_id')]);

        // Create notification
        createCollaborationNotification(collaborationId, 'ACCESS_REVOKED',
            collaboration[headers.indexOf('shared_with_user_id')],
            `Your access to ticket has been revoked: ${reason}`);

        // Log action
        logTicketAction(collaboration[headers.indexOf('ticket_id')], revokedByUserId, 'COLLABORATION_REVOKED', {
            collaboration_id: collaborationId,
            revocation_reason: reason
        });

        return createSuccessResponse({ message: 'Collaboration revoked successfully' });

    } catch (error) {
        return createErrorResponse("Failed to revoke collaboration: " + error.message);
    }
}

/**
 * Find tag assignment for ticket and tag
 */
function findTagAssignment(ticketId, tagId) {
    try {
        const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
        const sheet = spreadsheet.getSheetByName(TAG_SHEETS.TICKET_TAG_ASSIGNMENTS);

        if (!sheet) return null;

        const data = sheet.getDataRange().getValues();
        const headers = data[0];

        const assignmentRow = data.find(row =>
            row[headers.indexOf('ticket_id')] === ticketId &&
            row[headers.indexOf('tag_id')] === tagId &&
            row[headers.indexOf('is_active')] === true
        );

        if (assignmentRow) {
            const assignment = {};
            headers.forEach((header, index) => {
                assignment[header] = assignmentRow[index];
            });
            return assignment;
        }

        return null;
    } catch (error) {
        return null;
    }
}

/**
 * Increment tag usage count
 */
function incrementTagUsage(tagId, userId) {
    try {
        const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
        const tagSheet = spreadsheet.getSheetByName(TAG_SHEETS.TICKET_TAGS);
        const statsSheet = spreadsheet.getSheetByName(TAG_SHEETS.TAG_USAGE_STATISTICS);

        if (!tagSheet || !statsSheet) return;

        // Update tag usage count
        const tagData = tagSheet.getDataRange().getValues();
        const tagHeaders = tagData[0];
        const idColumnIndex = tagHeaders.indexOf('id');
        const usageColumnIndex = tagHeaders.indexOf('usage_count');

        const tagRowIndex = tagData.findIndex(row => row[idColumnIndex] === tagId);
        if (tagRowIndex !== -1) {
            const currentCount = tagData[tagRowIndex][usageColumnIndex] || 0;
            tagSheet.getRange(tagRowIndex + 1, usageColumnIndex + 1).setValue(currentCount + 1);
        }

        // Update user usage statistics
        const statsData = statsSheet.getDataRange().getValues();
        const statsHeaders = statsData[0];

        const existingStatRow = statsData.findIndex(row =>
            row[statsHeaders.indexOf('tag_id')] === tagId &&
            row[statsHeaders.indexOf('user_id')] === userId
        );

        if (existingStatRow !== -1) {
            // Update existing stat
            const currentCount = statsData[existingStatRow][statsHeaders.indexOf('usage_count')] || 0;
            statsSheet.getRange(existingStatRow + 1, statsHeaders.indexOf('usage_count') + 1).setValue(currentCount + 1);
            statsSheet.getRange(existingStatRow + 1, statsHeaders.indexOf('last_used_at') + 1).setValue(getCurrentTimestamp());
            statsSheet.getRange(existingStatRow + 1, statsHeaders.indexOf('updated_at') + 1).setValue(getCurrentTimestamp());
        } else {
            // Create new stat
            const statId = 'tag_stat_' + generateId();
            const newStat = {
                id: statId,
                tag_id: tagId,
                user_id: userId,
                company_id: null, // TODO: Get from user context
                usage_count: 1,
                last_used_at: getCurrentTimestamp(),
                created_at: getCurrentTimestamp(),
                updated_at: getCurrentTimestamp()
            };

            const newRow = statsHeaders.map(header => newStat[header] || '');
            statsSheet.appendRow(newRow);
        }

    } catch (error) {
        console.error('Error incrementing tag usage:', error);
    }
}

/**
 * =================================================================================
 * API ENDPOINT FUNCTIONS FOR TAGS AND COLLABORATION
 * Main API endpoint handlers that process HTTP requests
 * =================================================================================
 */

/**
 * Main API endpoint handler for tags and collaboration
 */
function handleTagsCollaborationAPI(action, data) {
    try {
        switch (action) {
            // Tag management endpoints
            case 'getTags':
                return getTicketTags(data.companyId, data.includeGlobal);
            case 'createTag':
                return createTicketTag(data);
            case 'updateTag':
                return updateTicketTag(data.tagId, data.updateData);
            case 'deleteTag':
                return deleteTicketTag(data.tagId, data.userId, data.reason);
            case 'assignTags':
                return assignTagsToTicket(data.ticketId, data.tagIds, data.assignedByUserId, data.notes);
            case 'removeTags':
                return removeTagsFromTicket(data.ticketId, data.tagIds, data.removedByUserId, data.reason);
            case 'getTicketTags':
                return getTicketTags(data.ticketId);
            case 'searchTicketsByTags':
                return searchTicketsByTags(data.tagIds, data.operator, data.companyId);

            // Collaboration endpoints
            case 'shareTicket':
                return shareTicket(data.ticketId, data.shareData);
            case 'requestAccess':
                return requestTicketAccess(data.ticketId, data.requestData);
            case 'respondToRequest':
                return respondToCollaborationRequest(data.requestId, data.responseData);
            case 'getSharedTickets':
                return getSharedTickets(data.userId, data.accessType);
            case 'revokeCollaboration':
                return revokeCollaboration(data.collaborationId, data.revokedByUserId, data.reason);

            default:
                return createErrorResponse("Unknown action: " + action);
        }
    } catch (error) {
        return createErrorResponse("API Error: " + error.message);
    }
}
